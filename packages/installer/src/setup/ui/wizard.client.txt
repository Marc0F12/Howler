  function chips(c) {
    if (S.checking) return `<span class="chip"><i></i>${t('c_check')}</span>`;
    if (!c) return '';
    if (!c.ok) return `<span class="chip bad"><i></i>${t('c_bad')}</span>` + c.problems.map((p) => `<span class="chip warn"><i></i>${esc(p)}</span>`).join('');
    return `<span class="chip ok"><i></i>${t('c_ok')}</span>`
      + `<span class="chip ${c.needsMigration ? 'warn' : 'ok'}"><i></i>${t(c.needsMigration ? 'c_up' : 'c_asv')}</span>`
      + (c.hasPlugin ? `<span class="chip ok"><i></i>${t('c_plug')}</span>` : '')
      + (c.presets[0] ? `<span class="chip"><i></i>${t('c_preset')}: ${esc(c.presets[0])}</span>` : '');
  }

  function server() {
    head(t('sv_title'), t('sv_sub'));
    const opt = (m, rec) => `<button type="button" class="opt" role="radio" aria-checked="${S.mode === m}" data-m="${m}"><span class="radio"></span><span><b>${t('m_' + m)}</b><span class="d">${t('m_' + m + '_d')}</span></span>${rec ? `<span class="tag">${t('rec')}</span>` : '<span></span>'}</button>`;
    const needDir = S.mode !== 'manual' && S.mode !== 'new';
    const found = S.init.servers.length
      ? `<div class="note" style="margin-top:14px">${t('found')}</div><div class="found">${S.init.servers.map((s, i) => `<button type="button" data-f="${i}" aria-pressed="${S.serverDir === s.dir}">${esc(s.dir)}</button>`).join('')}</div>`
      : `<p class="note" style="margin-top:14px">${t('none_found')}</p>`;
    const optOff = (m) => `<div class="opt off" aria-disabled="true"><span class="radio"></span><span><b>${t('m_' + m)}</b><span class="d">${t('new_need')}</span><button type="button" class="btn small" id="bac" style="margin-top:8px">${t('find_ac')}</button><span class="d" id="acerr" role="alert" style="color:var(--bad,#ff6b5e)">${esc(S.acErr || '')}</span></span><span></span></div>`;
    const hasServers = S.init.servers.length > 0;
    const hasAC = S.init.acDirs.length > 0;
    const newBox = S.mode === 'new' ? `<div class="note" style="margin-top:14px">${t('ac_found')}</div><div class="found">${S.init.acDirs.map((d, i) => `<button type="button" data-a="${i}" aria-pressed="${S.acDir === d}">${esc(d)}</button>`).join('')}</div>
      <div class="field"><label for="nd">${t('new_folder')}</label><div class="input"><input id="nd" value="${esc(S.newTo)}" spellcheck="false"><button class="btn small" type="button" id="b4">${t('browse')}</button></div></div>` : '';
    const noSrv = !hasServers && !hasAC && S.mode === 'manual' ? `<p class="note" style="margin-top:14px">${t('none_server')}</p>` : '';
    body(`<div class="opts" role="radiogroup">${hasServers ? opt('current', true) : ''}${hasServers ? opt('copy') : ''}${hasAC ? opt('new', !hasServers) : optOff('new')}${opt('manual', !hasServers && !hasAC)}</div>
      ${newBox}${noSrv}
      ${needDir && S.mode !== 'new' && (hasServers || S.mode !== 'manual') ? `<div class="field"><label for="sd">${t('srv_folder')}</label><div class="input"><input id="sd" value="${esc(S.serverDir)}" spellcheck="false"><button class="btn small" type="button" id="b1">${t('browse')}</button></div></div>
      <div id="chips">${chips(S.check)}</div>${S.serverDir ? '' : found}` : ''}
      ${S.mode === 'copy' ? `<div class="field"><label for="cd">${t('copy_folder')}</label><div class="input"><input id="cd" value="${esc(S.copyTo)}" spellcheck="false"><button class="btn small" type="button" id="b2">${t('browse')}</button></div></div>` : ''}`);
    document.querySelectorAll('[data-m]').forEach((b) => b.addEventListener('click', () => { S.mode = b.dataset.m; server(); }));
    document.querySelectorAll('[data-f]').forEach((b) => b.addEventListener('click', () => { S.serverDir = S.init.servers[+b.dataset.f].dir; checkServer(); server(); }));
    const bac = $('bac');
    if (bac) bac.addEventListener('click', async () => {
      const r = await api('browse', { initial: '' }).catch(() => null);
      if (!r || !r.dir) return;
      const v = await api('checkac', { dir: r.dir }).catch(() => ({ ok: false, problem: '' }));
      if (v.ok) { S.acErr = ''; if (!S.init.acDirs.includes(r.dir)) S.init.acDirs.push(r.dir); S.acDir = r.dir; S.mode = 'new'; if (!S.newTo) S.newTo = S.installDir + '\\server'; }
      else S.acErr = t('ac_bad') + ' ' + (v.problem || '');
      server();
    });
    document.querySelectorAll('[data-a]').forEach((b) => b.addEventListener('click', () => { S.acDir = S.init.acDirs[+b.dataset.a]; server(); }));
    const nd = $('nd');
    if (nd) {
      nd.addEventListener('input', () => { S.newTo = nd.value.trim(); gate(); });
      $('b4').addEventListener('click', async () => { const r = await api('browse', { initial: S.newTo }).catch(() => null); if (r && r.dir) { S.newTo = r.dir; server(); } });
    }
    const sd = $('sd');
    if (sd) {
      let tm;
      sd.addEventListener('input', () => { S.serverDir = sd.value.trim(); S.check = null; clearTimeout(tm); tm = setTimeout(checkServer, 450); gate(); });
      $('b1').addEventListener('click', async () => { const r = await api('browse', { initial: S.serverDir }).catch(() => null); if (r && r.dir) { S.serverDir = r.dir; checkServer(); server(); } });
    }
    const cd = $('cd');
    if (cd) {
      cd.addEventListener('input', () => { S.copyTo = cd.value.trim(); gate(); });
      $('b2').addEventListener('click', async () => { const r = await api('browse', { initial: S.copyTo }).catch(() => null); if (r && r.dir) { S.copyTo = r.dir; server(); } });
    }
    nav(t('back'), t('next'), { noNext: !serverReady() });
  }
  const serverReady = () => S.mode === 'manual' || (S.mode === 'new' && !!S.acDir && !!S.newTo) || (S.mode !== 'new' && !!S.check && S.check.ok && !S.checking && (S.mode !== 'copy' || !!S.copyTo));
  const gate = () => { $('next').disabled = !serverReady(); };
  async function checkServer() {
    if (!S.serverDir) { S.check = null; return; }
    S.checking = true; if ($('chips')) $('chips').innerHTML = chips(null);
    try { S.check = await api('check', { dir: S.serverDir }); } catch { S.check = null; }
    S.checking = false;
    if ($('chips')) $('chips').innerHTML = chips(S.check);
    gate();
  }

    else if (S.mode === 'new') L.push(`${t('p_newserver')} <code>${esc(S.newTo)}</code>`);
    else {
      const c = S.check || {};
      const target = S.mode === 'copy' ? S.copyTo : S.serverDir;
      if (S.mode === 'copy') L.push(`${t('p_copy')} <code>${esc(S.copyTo)}</code> — ${t('p_orig')}`);
      if (c.needsMigration) { L.push(`${t('p_backup')} <code>${esc(target)}\\howler-backup</code>`); L.push(t('p_migrate')); }
      else L.push(t('p_plugin'));
    }
    if (S.firewall) L.push(t('p_fw'));
    return L;
  }

  function options() {
    head(t('op_title'), t('op_sub'));
    const chk = (id, on, k) => `<div class="check" role="checkbox" tabindex="0" aria-checked="${on}" data-c="${id}"><span class="box">${ICONS.check}</span><span><b>${t(k)}</b><span class="d">${t(k + '_d')}</span></span></div>`;
    const itype = (k) => `<button type="button" class="opt" role="radio" aria-checked="${S.itype === k}" data-t="${k}"><span class="radio"></span><span><b>${t('it_' + k)}</b><span class="d">${t('it_' + k + '_d')}</span></span>${k === 'full' ? `<span class="tag">${t('rec')}</span>` : '<span></span>'}</button>`;
    body(`<div class="opts" role="radiogroup" aria-label="${t('it_title')}">${itype('full')}${itype('noports')}${itype('custom')}</div>
      <div class="field"><label for="id">${t('folder')}</label><div class="input"><input id="id" value="${esc(S.installDir)}" spellcheck="false"><button class="btn small" type="button" id="b3">${t('browse')}</button></div></div>
      ${S.itype === 'custom' ? `<div class="checks" style="margin-top:18px">${chk('firewall', S.firewall, 'fw')}${chk('shortcuts', S.shortcuts, 'sc')}${chk('autostart', S.autostart, 'au')}</div>` : ''}
      <div class="plan"><h2>${t('plan')}</h2><ul id="plan">${planList().map((x) => `<li>${x}</li>`).join('')}</ul></div>`);
    $('id').addEventListener('input', () => { S.installDir = $('id').value.trim(); $('plan').innerHTML = planList().map((x) => `<li>${x}</li>`).join(''); $('next').disabled = !S.installDir; });
    $('b3').addEventListener('click', async () => { const r = await api('browse', { initial: S.installDir }).catch(() => null); if (r && r.dir) { S.installDir = r.dir; options(); } });
    document.querySelectorAll('[data-t]').forEach((b) => b.addEventListener('click', () => {
      S.itype = b.dataset.t;
      if (S.itype === 'full') { S.firewall = true; S.shortcuts = true; S.autostart = false; }
      if (S.itype === 'noports') { S.firewall = false; S.shortcuts = true; S.autostart = false; }
      options();
    }));
    document.querySelectorAll('[data-c]').forEach((b) => {
      const flip = () => { S[b.dataset.c] = !S[b.dataset.c]; options(); };
      b.addEventListener('click', flip);
      b.addEventListener('keydown', (e) => { if (e.key === ' ' || e.key === 'Enter') { e.preventDefault(); flip(); } });
    });
    nav(t('back'), t(S.init.existing.exists ? 'update' : 'install'), { noNext: !S.installDir });
  }

  const TODO = ['check', 'files', 'copy', 'download', 'unpack', 'new', 'server', 'plugin', 'config', 'shortcuts', 'firewall', 'register'];
  function installing() {
    head(t('ins_title'), t('ins_sub'));
    const mig = S.mode !== 'manual' && S.mode !== 'new' && S.check && S.check.needsMigration;
    const used = (k) => ({ copy: S.mode === 'copy', download: (mig || S.mode === 'new') && !S.init.bundled, unpack: mig || S.mode === 'new', new: S.mode === 'new', server: mig, plugin: S.mode !== 'manual' && S.mode !== 'new' && S.check && !S.check.needsMigration, shortcuts: S.shortcuts || S.autostart, firewall: S.firewall })[k] ?? true;
    const list = TODO.filter(used);
    body(`<div class="bar"><i id="bar"></i></div><ul class="todo" id="todo">${list.map((k) => `<li data-k="${k}"><span class="ic"><span class="empty"></span></span><span>${t('t_' + k)}</span></li>`).join('')}</ul>`);
    nav('', '', {});
    S.shownTodo = list;
    start();
  }
  function paint(job) {
    const bar = $('bar'); if (bar) bar.style.width = Math.max(2, job.percent) + '%';
    const idx = S.shownTodo.indexOf(job.step);
    document.querySelectorAll('#todo li').forEach((li, i) => {
      const done = idx > i || job.state === 'done';
      li.className = done ? 'done' : i === idx ? 'on' : '';
      li.querySelector('.ic').innerHTML = done ? ICONS.check : i === idx ? '<span class="spin"></span>' : '<span class="empty"></span>';
    });
  }
  async function start() {
    try {
      await api('install', { mode: S.mode, serverDir: S.mode === 'manual' || S.mode === 'new' ? undefined : S.serverDir, copyTo: (S.mode === 'new' ? S.newTo : S.copyTo) || undefined, acDir: S.mode === 'new' ? S.acDir : undefined, installDir: S.installDir, firewall: S.firewall, shortcuts: S.shortcuts, autostart: S.autostart });
    } catch (e) { S.job = { state: 'error', error: e.message }; return finish(); }
    poll();
  }
  function poll() {
    clearInterval(S.timer);
    S.timer = setInterval(async () => {
      let job; try { job = await api('progress'); } catch { return; }
      S.job = job; paint(job);
      if (job.state === 'done' || job.state === 'error') { clearInterval(S.timer); setTimeout(finish, 350); }
    }, 300);
  }
  function finish() { S.step = 4; render(); }

  function done() {
    const j = S.job || {};
    if (j.state === 'error') {
      const known = ['RUNNING_HOWLER', 'RUNNING_SERVER'].includes(j.error);
      head(t('err_title'), t('err_sub'));
      body(`<div class="hero"><div class="big-ok bad">${ICONS.cross}</div></div>
        <div class="callout bad">${known ? t('e_' + j.error) : `${t('e_generic')} ${esc(j.error || '')}`}</div>
        ${known ? '' : `<details><summary class="note">${t('details')}</summary><pre>${esc(j.error || '')}</pre></details>`}`);
      nav(t('close'), t('retry'));
      $('back').onclick = () => quit();
      $('next').onclick = () => { S.step = 2; S.job = null; render(); };
      return;
    }
    const r = j.report || {};
    const up = S.init.existing.exists;
    head(t(up ? 'done_up' : 'done_title'), t('done_sub'));
    const notes = [];
    if (r.backupDir) notes.push(`<div class="callout ok">${t('d_backup')} <span class="mono">${esc(r.backupDir)}</span></div>`);
    if (S.mode === 'manual') notes.push(`<div class="callout">${t('d_manual')}</div>`);
    if ((r.warnings || []).includes('firewall-failed')) notes.push(`<div class="callout">${t('d_fw')}</div>`);
    if ((r.notes || []).includes('config-kept')) notes.push(`<div class="callout ok">${t('d_kept')}</div>`);
    body(`<div class="hero"><div class="big-ok">${ICONS.check}</div></div>${notes.join('')}`);
    nav(t('close'), t('open'));
    $('back').onclick = () => quit();
    $('next').onclick = async () => { await api('launch', {}).catch(() => undefined); quit(); };
  }

  /* ---- Desinstalar ---- */
  function uninstallPage() {
    $('steps').innerHTML = ''; $('version').textContent = 'v' + S.init.version;
    if (S.step === 4) {
      head(t('un_done'), '');
      body(S.deleteData ? '' : `<div class="callout ok">${t('un_kept')} <span class="mono">${esc(S.installDir)}</span></div>`);
      nav('', t('close'));
      $('next').onclick = () => quit();
      return;
    }
    if (S.job && S.job.state === 'error') {
      head(t('err_title'), '');
      body(`<div class="callout bad">${S.job.error === 'RUNNING_HOWLER' ? t('e_RUNNING_HOWLER') : esc(S.job.error)}</div>`);
      nav(t('close'), t('retry'));
      $('back').onclick = () => quit();
      $('next').onclick = () => { S.job = null; uninstallPage(); };
      return;
    }
    head(t('un_title'), t('un_sub'));
    body(`<div class="checks"><div class="check" role="checkbox" tabindex="0" aria-checked="${S.deleteData}" id="dd"><span class="box">${ICONS.check}</span><span><b>${t('un_data')}</b><span class="d">${t('un_data_d')}</span></span></div></div>`);
    const flip = () => { S.deleteData = !S.deleteData; uninstallPage(); };
    $('dd').addEventListener('click', flip);
    $('dd').addEventListener('keydown', (e) => { if (e.key === ' ' || e.key === 'Enter') { e.preventDefault(); flip(); } });
    nav(t('close'), t('un_btn'));
    $('back').onclick = () => quit();
    $('next').onclick = async () => {
      $('next').disabled = true; $('next').textContent = t('un_wait');
      try { await api('uninstall', { deleteData: S.deleteData }); } catch (e) { S.job = { state: 'error', error: e.message }; return uninstallPage(); }
      S.step = 4; uninstallPage();
    };
  }

  function modal(text) { $('modal-body').textContent = text; $('modal').hidden = false; $('modal-close').focus(); }
  $('modal-close').addEventListener('click', () => { $('modal').hidden = true; });
  document.addEventListener('keydown', (e) => { if (e.key === 'Escape') $('modal').hidden = true; });

  function render() {
    document.documentElement.lang = S.lang;
    if (S.init.mode === 'uninstall') return uninstallPage();
    rail();
    $('back').onclick = () => { if (S.step > 0 && S.step < 3) { S.step--; render(); } };
    $('next').onclick = () => {
      if (S.step === 1 && S.mode !== 'manual' && !serverReady()) return;
      if (S.step === 2) { S.step = 3; return render(); }
      if (S.step < 2) { S.step++; render(); }
    };
    [welcome, server, options, installing, done][S.step]();
  }

  // Latido: si se cierra la ventana, el servidor local se apaga solo.
  setInterval(() => api('ping', {}).catch(() => undefined), 2500);

  api('init').then((init) => {
    S.init = init; S.lang = init.lang === 'en' ? 'en' : 'es'; S.installDir = init.defaultDir;
    S.deleteData = false;
    if (init.servers.length) { S.serverDir = init.servers[0].dir; S.check = init.servers[0].info; }
    // Sin servidor detectado: se propone crear uno nuevo si hay Assetto Corsa, y si no, instalar solo Howler.
    else S.mode = init.acDirs.length ? 'new' : 'manual';
    if (init.acDirs.length) S.acDir = init.acDirs[0];
    S.newTo = init.defaultDir + '\\server';
    if (init.existing.exists && init.existing.serverDir) { S.serverDir = init.existing.serverDir; S.check = null; checkServer(); }
    render();
  }).catch((e) => { document.body.textContent = 'Error: ' + e.message; });
})();
